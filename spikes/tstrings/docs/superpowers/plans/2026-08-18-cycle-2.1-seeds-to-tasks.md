> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deterministically turn the 23 mined seeds into teachable `Task`s (one per seed per demonstrated operation), producing `tasks/built.jsonl`, `reports/build.md` (measured per-cell counts), and `reports/dropped.jsonl` (rejections with reasons).

**Architecture:** `types.py` gains `Provenance`, `Check`, `Task` plus `task_id()`/`semantic_id()` (canonical-json sha256). `cells.py` holds the 6-cell table (compose dropped) and deterministic operation analysis of a seed's AST. `build.py` extracts each seed's qualifying t-string literal + its simple assignments, derives a runnable reference per operation using canonical builders (the render walk inlined from the pinned source's `fstring`/`convert`), runs each reference in a lightweight subprocess to capture expected stdout, assigns a prompt from slot-filled families, and writes the outputs. Fully deterministic — no LLM (ground rule 2.1).

**Tech Stack:** Python 3.14 (spike-local pin), stdlib `ast`/`tomllib`/`json`/`hashlib`/`subprocess`, `click`, `pytest`, `ruff`. No third-party runtime deps.

```python
@dataclass(frozen=True, kw_only=True)
class Provenance:
    """Where a task came from, for audit and licensing."""

    source_id: str
    path: str
    line: int
    license: str


@dataclass(frozen=True, kw_only=True)
class Check:
    """One assertion a candidate program must satisfy."""

    kind: str  # "expected_value" | "expected_stdout" | "uses_feature"
    expected: str  # for uses_feature, the required module or node name


@dataclass(frozen=True, kw_only=True)
class Task:
    """One teachable unit: a prompt, a known-good solution, and its checks."""

    prompt: str
    reference: str
    checks: tuple[Check, ...]
    role: str
    operation: str
    provenance: Provenance
    task_id: str
    semantic_id: str


def _canonical_json(obj: object) -> str:
    """Return a canonical JSON string with sorted keys and compact separators."""
    return json.dumps(obj, sort_keys=True, separators=(",", ":"))


def _hash(fields: dict, *, exclude: frozenset[str]) -> str:
    """Return the sha256 of the canonical JSON of fields minus the excluded keys."""
    data = {k: v for k, v in fields.items() if k not in exclude}
    return hashlib.sha256(_canonical_json(data).encode()).hexdigest()


def task_id(fields: dict) -> str:
    """Return the sha256 of every field except task_id and semantic_id."""
    return _hash(fields, exclude=frozenset({"task_id", "semantic_id"}))


def semantic_id(fields: dict) -> str:
    """Return the sha256 of the same fields minus provenance."""
    return _hash(fields, exclude=frozenset({"task_id", "semantic_id", "provenance"}))
```

Create `src/satyrn/tstrings/cells.py`:

```python
"""Cell table and deterministic operation analysis for mined seeds."""

import ast
import tomllib
from pathlib import Path

CELLS: tuple[tuple[str, str], ...] = (
    ("author", "construct"),
    ("consumer", "read_strings"),
    ("consumer", "read_values"),
    ("consumer", "read_interpolations"),
    ("consumer", "render"),
    ("consumer", "negative_control"),
)


def load_cells(toml_path: Path) -> dict[tuple[str, str], int]:
    """Read min_tasks per cell from a cells.toml file."""
    with toml_path.open("rb") as fh:
        data = tomllib.load(fh)
    cells: dict[tuple[str, str], int] = {}
    for role, ops in data.get("cells", {}).items():
        for operation, entry in ops.items():
            cells[(role, operation)] = int(entry["min_tasks"])
    return cells


def _in_annotation_context(tree: ast.Module, nodes: list[ast.AST]) -> bool:
    """Return True if any TemplateStr sits in a signature annotation/default."""
    parent = {}
    for node in nodes:
        for child in ast.iter_child_nodes(node):
            parent[child] = node
    for node in nodes:
        if not isinstance(node, ast.TemplateStr):
            continue
        cur = node
        while cur in parent:
            cur = parent[cur]
            if isinstance(cur, (ast.arg, ast.AnnAssign)):
                return True
    return False


def operations_of(seed_text: str) -> set[str]:
    """Return the operations a seed's text demonstrates, deterministically."""
    tree = ast.parse(seed_text)
    nodes = list(ast.walk(tree))
    if not any(isinstance(n, ast.TemplateStr) for n in nodes):
        return set()
    if _in_annotation_context(tree, nodes):
        return set()
    attrs = {n.attr for n in nodes if isinstance(n, ast.Attribute)}
    calls = {
        n.func.id
        for n in nodes
        if isinstance(n, ast.Call) and isinstance(n.func, ast.Name)
    }
    fstrings = [n for n in nodes if isinstance(n, ast.JoinedStr) and not isinstance(n, ast.TemplateStr)]

    ops: set[str] = {"construct"}
    if "interpolations" in attrs or "conversion" in attrs or "format_spec" in attrs:
        ops.add("read_interpolations")
    if "values" in attrs:
        ops.add("read_values")
    if "fstring" in calls or ("strings" in attrs and "values" in attrs):
        ops.add("render")
        ops.add("read_strings")
        ops.add("read_values")
    elif "strings" in attrs:
        ops.add("read_strings")
    if fstrings:
        ops.add("negative_control")
    return ops
```

The annotation rule walks up from each `TemplateStr`; hitting an `ast.arg` means the t-string sits in a function signature (annotation or default — e.g. `def f(x: t"{a}")`) and hitting `ast.AnnAssign` means a variable annotation. Both return the empty set, which is what `test_operations_of_annotation_seed_is_empty` enforces.

## Task 3: Reference derivation, execution, and checks

**Files:**
- Create: `src/satyrn/tstrings/build.py` (reference-building part)
- Create: `tests/test_build.py` (reference/execution tests)

**Interfaces:**
- Consumes: `Task`, `Check`, `Provenance` (Task 1); `operations_of` (Task 2).
- Produces:
  - `extract_literal(seed_text: str, need_interpolation: bool) -> tuple[str, list[str]] | None` — returns `(literal_source, assignment_lines)` for the first t-string literal that satisfies `need_interpolation` and whose interpolations reference only names bound by simple `NAME = <constant>` assignments earlier in the seed (or are constant expressions); `None` if no literal qualifies.
  - `build_reference(operation: str, seed_text: str) -> tuple[str, tuple[Check, ...]] | None` — a complete runnable program + checks (includes a `uses_feature` check for every non-negative_control operation).
  - `run_reference(reference: str, timeout: int = 10) -> tuple[int, str]` — subprocess run, returns `(returncode, stdout)`.

**Extraction rule (exact):** scan the seed's module statements in order; for each `ast.Assign` whose value is (or contains at its top level) a `TemplateStr`, record the literal source via `ast.get_source_segment` and its referenced interpolation names (each `ast.FormattedValue` with an `ast.Name` value). A literal qualifies when (a) `need_interpolation` is False, or the literal has ≥1 interpolation; and (b) every referenced name has a preceding simple assignment `NAME = <ast.Constant>` in the seed. If the first literal fails (b), try the next. Simple assignments are collected as their full source lines. Constant-expression interpolations (`{1}`, `{1 / 0.3:.2f}`) need no assignment and are allowed.

**Reference builders (canonical, exact):**

- `construct`:
  ```python
  from string.templatelib import Template

  {assignments}
  t = {literal}
  print(type(t).__name__)
  ```
  Expected stdout: `Template`. `uses_feature` check: `("uses_feature", "string.templatelib")`.

- `render` — the render walk is inlined from the pinned source's `fstring`/`convert` (`Lib/test/test_string/_support.py`):
  ```python
  from string.templatelib import Template, Interpolation

  def _convert(value, conversion):
      if conversion == "a":
          return ascii(value)
      if conversion == "r":
          return repr(value)
      if conversion == "s":
          return str(value)
      return value

  def _render(template):
      parts = []
      for item in template:
          if isinstance(item, str):
              parts.append(item)
          elif isinstance(item, Interpolation):
              parts.append(format(_convert(item.value, item.conversion), item.format_spec))
      return "".join(parts)

  {assignments}
  t = {literal}
  print(_render(t))
  ```

- `read_strings`:
  ```python
  from string.templatelib import Template

  {assignments}
  t = {literal}
  print(t.strings)
  ```

- `read_values`:
  ```python
  from string.templatelib import Template

  {assignments}
  t = {literal}
  print(t.values)
  ```

- `read_interpolations`:
  ```python
  from string.templatelib import Template

  {assignments}
  t = {literal}
  print(t.interpolations[0].expression)
  ```
  (requires the chosen literal to have ≥1 interpolation)

- `negative_control`: take the seed's first f-string expression, extract its `JoinedStr` source and the preceding simple assignments its names reference, and emit:
  ```python
  {assignments}
  result = {fstring_source}
  print(result)
  ```
  No `uses_feature` check — instead a `Check(kind="expected_stdout", ...)` and a negative-usage guarantee (the reference contains no `TemplateStr`, asserted in tests).

**Execution:** `run_reference` runs `[sys.executable, "-c", reference]` with `subprocess.run(..., capture_output=True, text=True, timeout=timeout)`. On nonzero return, timeout, or non-`Template`-type output for construct — the candidate is dropped with reason (Task 5 records it). `expected_stdout` is the captured stdout.

```python
"""Deterministic task building from mined seeds."""

import ast
import subprocess
import sys
from pathlib import Path

from satyrn.tstrings.types import Check

_CONSTRUCT_HEADER = "from string.templatelib import Template\n\n"
_RENDER_HEADER = (
    "from string.templatelib import Template, Interpolation\n\n"
    "def _convert(value, conversion):\n"
    '    if conversion == "a":\n'
    "        return ascii(value)\n"
    '    if conversion == "r":\n'
    "        return repr(value)\n"
    '    if conversion == "s":\n'
    "        return str(value)\n"
    "    return value\n\n"
    "def _render(template):\n"
    "    parts = []\n"
    "    for item in template:\n"
    "        if isinstance(item, str):\n"
    "            parts.append(item)\n"
    "        elif isinstance(item, Interpolation):\n"
    "            parts.append(format(_convert(item.value, item.conversion), item.format_spec))\n"
    '    return "".join(parts)\n\n'
)


def extract_literal(seed_text: str, need_interpolation: bool) -> tuple[str, list[str]] | None:
    """Return (literal_source, assignment_lines) for the first qualifying literal."""
    tree = ast.parse(seed_text)
    assignments: dict[str, str] = {}
    parent = {}
    for node in ast.walk(tree):
        for child in ast.iter_child_nodes(node):
            parent[child] = node
    for node in ast.walk(tree):
        if isinstance(node, ast.Assign) and len(node.targets) == 1:
            if isinstance(node.targets[0], ast.Name) and isinstance(node.value, ast.Constant):
                assignments[node.targets[0].id] = ast.get_source_segment(seed_text, node)
    for node in ast.walk(tree):
        if not isinstance(node, ast.TemplateStr):
            continue
        interps = [n for n in ast.walk(node) if isinstance(n, ast.FormattedValue)]
        if need_interpolation and not interps:
            continue
        names = {n.id for n in interps if isinstance(n.value, ast.Name)}
        if names - set(assignments):
            continue
        literal = ast.get_source_segment(seed_text, node)
        if literal is None:
            continue
        needed = [assignments[n] for n in names if n in assignments]
        return literal, needed
    return None
```

**Interfaces:**
- Consumes: nothing new.
- Produces: `PROMPT_FAMILIES: dict[str, tuple[str, ...]]` — six families, each a tuple of phrasings; `fill_family(family: str, operation: str, seed_values: dict) -> str`; `prompt_fingerprint(prompt: str, seed_tokens: set[str]) -> str`.

**Fingerprint rule (exact, BRIEF Phase 2):** `prompt_fingerprint` lowercases the prompt, collapses runs of whitespace to one space, and replaces every token drawn from the seed (identifiers and literal values present in the seed text) with `#`. Two prompts from the same family+operation with different variable names must produce the same fingerprint.

**Families (exact — six families, each ≥1 phrasing; the build fills `{op}` and seed content):**

| Family | Phrasing(s) |
|---|---|
| `build` | "Using Python t-strings, {op}. Print the result." |
| `create` | "Write a Python program that {op}. Print what it produces." |
| `show` | "Demonstrate Python t-strings by {op}. Print the outcome." |
| `teach` | "Show how Python t-strings let you {op}. Print the value." |
| `example` | "Give an example where Python t-strings {op}. Print the result." |
| `output` | "Produce a small Python example that {op}. Print the output." |

where `{op}` is a per-operation verb phrase: construct → `"build a Template"`; render → `"render a Template to a string"`; read_strings → `"extract a Template's static parts (its .strings tuple)"`; read_values → `"read a Template's evaluated values (its .values tuple)"`; read_interpolations → `"inspect an interpolation's expression, conversion, and format spec"`; negative_control → `"format a value into a string using ordinary Python string features (no t-strings)"`.

**`build_tasks` algorithm (exact):**
1. Load seeds (JSONL → `Seed`). Load cells.
2. For each seed: `ops = operations_of(seed.text)`; drop to `reports/dropped.jsonl` with `reason="no clean operation"` if empty.
3. For each operation in a deterministic order (construct, render, read_strings, read_values, read_interpolations, negative_control): if the operation needs an interpolation and none qualifies, or `build_reference` returns None, record the drop with `reason="<op>: no qualifying literal"` and continue. Else `(reference, checks)`; run `run_reference`; on nonzero/timeout record `reason="<op>: reference failed to run"`; else append the expected_stdout check and build the `Task`:
   - `role`/`operation` from the cell;
   - `prompt` = `fill_family(family_for(task), operation, {})` — rotate families across tasks so every occupied cell uses ≥4 distinct families and the total distinct `(family, operation)` fingerprints reach the measured target (report the count in `reports/build.md`);
   - `provenance` from the seed (`source_id`, `path`, `line`, `license="PSF-2.0"`);
   - `task_id`/`semantic_id` via Task 1.
4. Write `tasks/built.jsonl` (one JSON `Task` per line, via `asdict`), `reports/build.md` (per-cell counts + total distinct fingerprint count + dropped counts), and `reports/dropped.jsonl` (each with `reason`).

**Family rotation rule (exact):** iterate families round-robin over the tasks in cell order; a cell with ≥4 tasks necessarily receives ≥4 distinct families. `family_for` = next family in rotation for that cell.

```python
@click.command("build")
@click.option("-i", "--input", type=click.Path(exists=True, path_type=Path), required=True, help="Mined seeds JSONL to build tasks from.")
@click.option("-o", "--output-dir", type=click.Path(file_okay=False, path_type=Path), default="tasks", show_default=True, help="Directory to write built tasks into.")
def main(input: Path, output_dir: Path) -> None:
    """Build teachable tasks from mined seeds."""
    spike_root = Path(__file__).resolve().parents[3]
    tasks = build_tasks(input, spike_root / "cells.toml", output_dir, spike_root / "reports")
    output_path = output_dir / "built.jsonl"
    output_path.parent.mkdir(parents=True, exist_ok=True)
    with output_path.open("w") as fh:
        for task in tasks:
            fh.write(json.dumps(asdict(task)) + "\n")
    click.echo(f"Wrote {len(tasks)} tasks to {output_path}")
```
