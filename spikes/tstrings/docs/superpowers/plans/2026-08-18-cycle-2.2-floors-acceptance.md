**Return annotation (`src/satyrn/tstrings/cells.py`)** — in `_in_annotation_context`, when the walk-up reaches a `FunctionDef`/`AsyncFunctionDef`, also return True if the TemplateStr is within that function's `.returns` subtree:

```python
    for node in nodes:
        if not isinstance(node, ast.TemplateStr):
            continue
        cur = node
        while cur in parent:
            cur = parent[cur]
            if isinstance(cur, (ast.arg, ast.AnnAssign)):
                return True
            if isinstance(cur, (ast.FunctionDef, ast.AsyncFunctionDef)):
                if node in ast.walk(cur.returns) if cur.returns else False:
                    return True
    return False
```

(If `cur.returns` is None, the template is in the body — not annotation context.)

**Preceding-only assignments (`src/satyrn/tstrings/build.py`)** — `_collect_simple_assignments` must record each assignment's end position, and `extract_literal` must only accept assignments that strictly precede the literal's start position:

```python
def _collect_simple_assignments(tree: ast.Module, source: str) -> dict[str, tuple[str, int, int]]:
    """Return NAME -> (assignment_source, end_lineno, end_col) for NAME=<constant>."""
    out: dict[str, tuple[str, int, int]] = {}
    for node in ast.walk(tree):
        if not isinstance(node, ast.Assign) or len(node.targets) != 1:
            continue
        target = node.targets[0]
        if isinstance(target, ast.Name) and isinstance(node.value, ast.Constant):
            segment = ast.get_source_segment(source, node)
            if segment is not None:
                out[target.id] = (segment, node.end_lineno or node.lineno, node.end_col_offset)
    return out
```

In `extract_literal`, replace `names - set(assignments)` with a positional check (the literal's `(lineno, col_offset)` must be strictly after the assignment's `(end_lineno, end_col_offset)`):

```python
        literal_pos = (node.lineno, node.col_offset)
        missing = [
            name
            for name in names
            if name not in assignments
            or (assignments[name][1], assignments[name][2]) >= literal_pos
        ]
        if missing:
            continue
        needed = [assignments[name][0] for name in sorted(names)]
```

FINGERPRINT_FLOOR = 26  # updated to the re-measured value in Step 2


def test_every_cell_meets_its_floor() -> None:
    """The real build on the 23 seeds meets every cells.toml floor."""
    from satyrn.tstrings.build import build_tasks
    from satyrn.tstrings.cells import load_cells

    spike_root = Path(__file__).resolve().parents[1]
    tasks = build_tasks(spike_root / "seeds" / "mined.jsonl", spike_root / "cells.toml", None, None)
    floors = load_cells(spike_root / "cells.toml")
    counts: dict[tuple[str, str], int] = {}
    for t in tasks:
        counts[(t.role, t.operation)] = counts.get((t.role, t.operation), 0) + 1
    for cell, floor in floors.items():
        assert counts.get(cell, 0) >= floor, f"cell {cell} below floor"


def test_fingerprint_floor_met() -> None:
    """The committed build report's distinct fingerprints meet the floor."""
    import re as _re

    spike_root = Path(__file__).resolve().parents[1]
    report = (spike_root / "reports" / "build.md").read_text()
    match = _re.search(r"Distinct \(family, operation\) fingerprints: (\d+)", report)
    assert match is not None
    assert int(match.group(1)) >= FINGERPRINT_FLOOR
